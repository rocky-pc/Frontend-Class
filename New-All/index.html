var vs let vs const

1. Declaration

var firstName;
      console.log(firstName);

      let lastName;
      console.log(lastName);

      const fullName;
      console.log(fullName);

2. Re-Assign Value

var firstName = 'JS';
      firstName = 'JavaScript';
      console.log(firstName);

      let lastName = 'ES6';
      lastName = 'EcmaScript6+';
      console.log(lastName);

      const fullName = 'JavaScript';
      fullName = 'JS ES6+';
      console.log(fullName);

3. Re-Declaration

      var firstName = 'Java';
      var firstName = 'JS';
      console.log(firstName);

      let lastName = 'Script';
      let lastName = 'JavaScript';
      console.log(lastName);

      const fullName = 'JavaScript';
      const fullName = 'JavaScript ES6+';
      console.log(fullName);

Switch Case

Syntax

switch(expression) {
  case value1:
    // code block
    break;
  case value2:
    // code block
    break;
  default:
    // default code block
}

Example: Student Grade

function getStudentGrade(rank) {
  switch(rank) {
    case 'A':
      return 'Good Student';
    case 'B':
      return 'Average Student';
    case 'C':
      return 'Poor Student';
    default:
      return 'Legend';
  }
}

// Demo values
console.log('Rank A:', getStudentGrade('A'));
console.log('Rank B:', getStudentGrade('B'));
console.log('Rank C:', getStudentGrade('C'));
console.log('Rank X:', getStudentGrade('X'));

Loops

For Loop

// Basic for loop
for(let i = 1; i <= 10; i++){
  console.log(i);
}

// Loop through array
let colors = ['red','green','blue','yellow','orange','black','white'];

for(let i = 0; i < colors.length; i++){
  console.log(i + 1, colors[i]);
}

Nested Loop

for(let i = 1; i <= 5; i++){
  console.log(i, 'Outer Loop');

  for(let j = 1; j <= 3; j++ ){
    console.log(j, 'Inner Loop');
  }
}

// Break and Continue
for(let j = 1; j <= 6; j++ ){
  if(j == 4){
    break; // or continue;
  }
  console.log(j);
}

While Loop

let i = 0;
while(i < 5){
  console.log(i);
  i++;
}

Do While Loop

let j = 0;
do{
  console.log(j);
  j++;
}while(j < 10);

Scopes in JavaScript

Scope defines the context in which variables are accessible. It determines the lifetime and visibility of variables.

Global Scope

Variables declared outside any function or block are accessible anywhere in the code.

let data = 'Hi';

      // Accessible inside block
      {
        console.log(data);
      }

      // Accessible outside block
      console.log(data);

Block Scope

Introduced with let and const in ES6, variables declared inside {} are accessible only within that block.

// This will throw an error
      {
        let message = 'Good Morning!';
      }

      console.log(message); // Error: message is not defined

Hoisting

Hoisting is JavaScript's default behavior of moving declarations to the top of the current scope.

Example

console.log(text); // undefined (not an error)

let text = 'Hello Guys';

Note: The variable is hoisted but the value is not. It returns undefined until the assignment is reached.
